\section{Policy Network and Training Algorithm}
\label{sec:policy-network}
% -----------------------------------------------------------

% TODO ¶1 [~150 w] Architecture
% ResNet-18 feature extractor (paper-faithful, ImageNet-init).
% Input channels: current reconstruction x_t plus auxiliary maps
% (iteration index broadcast, noise-level broadcast) = 3 channels.
% Three output heads:
%   - pi_1 (termination): Bernoulli logit
%   - pi_2 (parameters): two continuous outputs squashed via
%       sigmoid to (sigma_k in [1,5], mu_k in [10,100])
%   - V (critic): scalar value
% Total ~11M parameters.
%
% Insert Fig 4.5 here (policy network architecture).

% TODO ¶2 [~150 w] A2C-style objective
% Three loss terms:
%   - L_critic = (V(s_t) - R_t)^2     (Wei2022 eq. 15)
%   - L_pi1 = -log pi_1(a_t|s_t) * A_t    (REINFORCE)
%   - L_pi2 = -R_t                        (reparameterised through
%                                          the differentiable env)
% A_t = R_t - V(s_t) is the advantage.
% pi_2 warmup: critic trains for 5 epochs while pi_2 stays at
% initialisation. Without warmup the critic baseline is meaningless
% and pi_2 gradients are catastrophically noisy.

% TODO ¶3 [~100 w] Replay buffer
% Episodes pushed to a CPU-side replay buffer, sampled in
% mini-batches of 8 transitions for gradient updates. Each
% transition ~6 MB (two 512x512 float32 maps + scalars). Buffer
% size = 5,000 (48G RAM headroom on A100 nodes) or 10,000 (full
% dataset, 80G nodes). FIFO eviction.
